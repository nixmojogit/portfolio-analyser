python - << 'PYEOF'
import sqlite3
from pathlib import Path

db = Path("data/portfolio/portfolio.db")
with sqlite3.connect(db) as conn:
    # Clear all related tables for a clean re-import
    conn.execute("DELETE FROM holdings")
    conn.execute("DELETE FROM scores_history")
    conn.execute("DELETE FROM recommendations_history")
    conn.execute("DELETE FROM alerts_log")
    conn.commit()
    
    # Verify
    count = conn.execute("SELECT COUNT(*) FROM holdings").fetchone()[0]
    print(f"Holdings remaining: {count}")
    print("Database cleared successfully")
PYEOF
